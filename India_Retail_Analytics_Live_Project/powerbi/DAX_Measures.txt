Total Sales = SUM(Orders[Sales])
Total Profit = SUM(Orders[Profit])
Orders Count = DISTINCTCOUNT(Orders[Order_ID])
Units Sold = SUM(Orders[Quantity])
Average Order Value = DIVIDE([Total Sales],[Orders Count])
Profit Margin % = DIVIDE([Total Profit],[Total Sales])
Return Rate % = DIVIDE(SUM(Orders[Returned]),[Orders Count])
On Time Delivery % = DIVIDE(CALCULATE([Orders Count],Orders[Delivery_Status]="On Time"),[Orders Count])
Late Orders = CALCULATE([Orders Count],Orders[Delivery_Status]="Late")
Average Rating = AVERAGE(Orders[Customer_Rating])
Sales MoM % = VAR CurrentSales=[Total Sales] VAR PreviousSales=CALCULATE([Total Sales],DATEADD('Date'[Date],-1,MONTH)) RETURN DIVIDE(CurrentSales-PreviousSales,PreviousSales)
Sales YTD = TOTALYTD([Total Sales],'Date'[Date])
Profit YTD = TOTALYTD([Total Profit],'Date'[Date])
